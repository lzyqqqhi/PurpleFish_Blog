      # Vue3 compiler-core 源码

      一句话：把模板字符串 → AST → render 函数字符串。

      阶段 函数 职责
      解析 Parse baseParse template → AST
      转换 Transform transform AST 遍历，处理指令、{{}}、v-if…
      生成 Codegen generate AST → 可执行 render 字符串
      • 伪代码
      // compile.ts
// src/index.ts
export { baseCompile } from "./compile";

// src/compiler.ts
import { generate } from "./codegen";
import { baseParse } from "./parse";
import { transform } from "./transform";
import { transformExpression } from "./transforms/transformExpression";
import { transformElement } from "./transforms/transformElement";
import { transformText } from "./transforms/transformText";

export function baseCompile(template, options) {
  // 1. 先把 template 也就是字符串 parse 成 ast
  const ast = baseParse(template);
  // 2. 给 ast 加点料（- -#）
  transform(
    ast,
    Object.assign(options, {
      nodeTransforms: [transformElement, transformText, transformExpression],
    })
  );
  
  // 3. 生成 render 函数代码
  return generate(ast);
}

      # Parse：有限状态机（FSM）

      当前状态 触发字符 下一步动作 生成的 AST 节点
      初始 {{ 进入“插值”状态 INTERPOLATION
      初始 < 进入“标签”状态 ELEMENT（开始标签）
      标签内 </ 进入“结束标签”状态 结束标签（不产生新节点，仅闭合）
      其他 任意文本 进入“文本”状态 TEXT

      • AST结构

        interface ElementNode {
  type: NodeTypes.ELEMENT;        // 固定为 NodeTypes.ELEMENT（一个枚举值）
  tag: string;                   // 标签名，例如 "div"、"span"
  props: Array<AttributeNode | DirectiveNode>; // 属性/指令列表
  children: TemplateChildNode[]; // 子节点（可以是元素、文本、插值、注释等）
  isSelfClosing: boolean;        // 是否是自闭合标签，如 <img/>
  loc: SourceLocation;           // 在源码中的起止位置（调试/报错用）
}

        interface TextNode {
  type: NodeTypes.TEXT;   // 固定为 NodeTypes.TEXT
  content: string;        // 文本内容本身，如 "hello"
  loc: SourceLocation;    // 源码中的起止位置（调试/报错用）
}

      # tranform

      # 1 阶段定位

      在 parse 之后、generate 之前。 输入：纯语法树（节点只有 tag、props、children）。 输出：同样一棵树，但每个节点多了

      • 处理后的指令（v-if → 三元表达式）
      • 静态提升标记（PatchFlag）
      • helpers 清单（运行时要用到的 createVNode、toDisplayString…）

      # 2 官方三大内置插件

      插件文件 进入阶段做的事 退出阶段做的事
      transformElement 1. 解析 v-bind、v-on 等指令 生成 codegenNode
      transformExpression 把 中的 msg 转成 _ctx.msg —（无 exit）
      transformText 合并相邻文本节点 & 插值节点 生成 createTextVNode("hello " + _toDisplayString(_ctx.msg))
      export function transform(root, options = {}) {
  // 1. 创建 context
  
  const context = createTransformContext(root, options);
  
  // 2. 遍历 node
  traverseNode(root, context);
  
  createRootCodegen(root, context);
  
  root.helpers.push(...context.helpers.keys());
}

function createTransformContext(root, options): any {
  const context = {
    root,
    nodeTransforms: options.nodeTransforms || [],
    helpers: new Map(),
    helper(name) {
      // 这里会收集调用的次数
      // 收集次数是为了给删除做处理的， （当只有 count 为0 的时候才需要真的删除掉）
      // helpers 数据会在后续生成代码的时候用到
      const count = context.helpers.get(name) || 0;
      context.helpers.set(name, count + 1);
    },
  };
  
  return context;
}

function traverseNode(node: any, context) {
  const type: NodeTypes = node.type;
  
  // 遍历调用所有的 nodeTransforms
  // 把 node 给到 transform
  // 用户可以对 node 做处理
  const nodeTransforms = context.nodeTransforms;
  const exitFns: any = [];
  for (let i = 0; i < nodeTransforms.length; i++) {
    const transform = nodeTransforms[i];
    
    const onExit = transform(node, context);
    if (onExit) {
      exitFns.push(onExit);
    }
  }
  
  switch (type) {
    case NodeTypes.INTERPOLATION:
      // 插值的点，在于后续生成 render 代码的时候是获取变量的值
      context.helper(TO_DISPLAY_STRING);
      break;
      
    case NodeTypes.ROOT:
    case NodeTypes.ELEMENT:
      
      traverseChildren(node, context);
      break;
      
    default:
      break;
  }
  
  
  
  let i = exitFns.length;
  // i-- 这个很巧妙
  // 使用 while 是要比 for 快 (可以使用 https://jsbench.me/ 来测试一下)
  while (i--) {
    exitFns[i]();
  }
}

function createRootCodegen(root: any, context: any) {
  const { children } = root;
  
  // 只支持有一个根节点
  // 并且还是一个 single text node
  const child = children[0];
  
  // 如果是 element 类型的话 ， 那么我们需要把它的 codegenNode 赋值给 root
  // root 其实是个空的什么数据都没有的节点
  // 所以这里需要额外的处理 codegenNode
  // codegenNode 的目的是专门为了 codegen 准备的  为的就是和 ast 的 node 分离开
  if (child.type === NodeTypes.ELEMENT && child.codegenNode) {
    const codegenNode = child.codegenNode;
    root.codegenNode = codegenNode;
  } else {
    root.codegenNode = child;
  }
}

      # 总流程

      1. **解析（Parse）**模板字符串（template string） → 生成 AST关键函数：baseParse
      2. **转换（Transform）**遍历 AST，应用指令、计算属性等节点转换关键函数 v- {{：transform
      3. **生成代码（Codegen）**把转换后的 AST 转为可被执行的渲染代码关键函数：generate
