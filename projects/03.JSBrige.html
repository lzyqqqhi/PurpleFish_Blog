      # JSBridge封装

      # 一、源码封装（已加中文注释）

      class JSBridge {
  constructor() {
    this.callbackId = 0;          // 回调序号，保证唯一
    this.callbacks = {};          // 保存网页发起的回调
    this.handlers = {};           // 保存供原生调用的函数
    this.initialized = false;     // 防止重复初始化
    this.init();                  // 立即初始化
  }

  /* 1. 初始化：暴露全局对象给原生，发事件通知网页已就绪 */
  init() {
    if (this.initialized || typeof window === 'undefined') return;

    window.JSBridge = {           // 原生通过它调网页
      invoke: (handlerName, data, callbackId) =>
        this.invokeHandler(handlerName, data, callbackId),
      callback: (callbackId, data) =>
        this.invokeCallback(callbackId, data),
    };

    this.dispatchEvent('JSBridgeReady'); // 网页可监听此事件
    this.initialized = true;
  }

  /* 2. 网页注册函数，供原生反向调用 */
  registerHandler(handlerName, handler) {
    if (typeof handler !== 'function') return;
    this.handlers[handlerName] = handler;
  }

  /* 3. 网页调原生（最常用）*/
  callHandler(handlerName, data = {}, callback?) {
    if (!this.initialized) return;

    // 生成唯一回调 ID
    const callbackId = callback ? `cb_${++this.callbackId}_${Date.now()}` : null;
    if (callbackId) this.callbacks[callbackId] = callback;

    // 拼消息并发送
    const message = { handlerName, data, callbackId };
    this.dispatchToNative(message);
  }

  /* 4. 原生调网页（内部入口）*/
  invokeHandler(handlerName, data, callbackId) {
    const handler = this.handlers[handlerName];
    if (!handler) {
      this.callbackToNative(callbackId, { error: '方法未注册' });
      return;
    }
    try {
      const result = handler(data);
      if (callbackId) this.callbackToNative(callbackId, result);
    } catch (e) {
      if (callbackId) this.callbackToNative(callbackId, { error: e.message });
    }
  }

  /* 5. 原生回传结果（内部）*/
  invokeCallback(callbackId, data) {
    const callback = this.callbacks[callbackId];
    if (callback) {
      callback(data);
      delete this.callbacks[callbackId]; // 一次性
    }
  }

  /* 6. 把结果再发回原生（内部）*/
  callbackToNative(callbackId, data) {
    this.dispatchToNative({ callbackId, data });
  }

  /* 7. 真正发送消息到原生（平台差异）*/
  dispatchToNative(message) {
    try {
      const msgStr = JSON.stringify(message);
      if (this.isAndroid() && window.AndroidJSBridge) {
        window.AndroidJSBridge.invoke(msgStr);          // Android 注入
      } else if (this.isIOS()) {
        const iframe = document.createElement('iframe'); // iOS 拦截
        iframe.style.display = 'none';
        iframe.src = `jsbridge://message?data=${encodeURIComponent(msgStr)}`;
        document.body.appendChild(iframe);
        setTimeout(() => iframe.remove(), 0);
      } else {
        // 非 App 环境：1 秒后模拟返回（开发调试）
        asyncClientOperation(msgStr, (err, res) => {
          if (message.callbackId) this.invokeCallback(message.callbackId, err ? { error: err } : res);
        });
      }
    } catch (e) {
      console.error('JSBridge: 发送失败', e);
    }
  }

  /* 8. 工具：发事件 + 平台判断 */
  dispatchEvent(eventName, data = null) {
    window.dispatchEvent(new CustomEvent(eventName, { detail: data }));
  }
  isAndroid() { return /Android/i.test(navigator.userAgent); }
  isIOS() { return /iPhone|iPad|iPod|iOS/i.test(navigator.userAgent); }
}

/* 9. 单例导出 + 工具函数 */
const jsBridge = new JSBridge();
export default jsBridge;

/* 等待初始化完成再执行业务 */
export function readyJSBridge(cb: Function) {
  if (jsBridge.initialized) cb(jsBridge);
  else window.addEventListener('JSBridgeReady', () => cb(jsBridge), { once: true });
}

      # 二、使用示例

      # 网页调原生支付（H5 → App）

      import jsBridge from './JSBridge';

async function pay() {
  // 1. 调原生收银台，等待结果
  const res = await new Promise<any>((resolve) => {
    jsBridge.callHandler('openCashier', { price: 100 }, resolve);
  });

  // 2. 根据结果跳转
  if (res.status === 'paySucc') {
    location.href = '/pay-success';
  } else {
    alert('支付失败：' + res.message);
  }
}

// 绑定按钮
document.getElementById('payBtn')!.onclick = pay;

      # 原生调网页抽奖（App → H5）

      // 1. 注册一个供原生调用的函数
jsBridge.registerHandler('lottery', (data, callbackId) => {
  console.log('[H5] 收到原生参数：', data);
  const prize = Math.random() > 0.5 ? 'iPhone' : '优惠券';

  // 2. 把结果回传原生（原生会收到 Promise resolve）
  return { prize, level: data.userLevel };
});

      # 三、全流程

      sequenceDiagram
  participant W as 网页
  participant B as JSBridge
  participant A as 原生App

  W->>B: callHandler('openCashier',data,cb)
  B->>A: AndroidJSBridge.invoke 或 iframe
  A->>A: 打开收银台
  A->>B: window.JSBridge.callback(id,result)
  B->>W: cb(result) → Promise resolve
