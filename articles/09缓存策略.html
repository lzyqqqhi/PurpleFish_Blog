# Vue3 + IndexedDB 版 SWR（Stale-While-Revalidate）自定义 Hook

// hooks/useSWR.ts
import { reactive, toRefs, watchEffect } from 'vue'
import { get as idbGet, set as idbSet } from 'idb-keyval'

export default function useSWR(key: string, fetcher: () => Promise<any>, opts = {}) {
  const state = reactive({ data: null, error: null, loading: false })
  const { refreshInterval = 0, ttl = 60000 } = opts

  async function revalidate() {
    state.loading = true
    try {
      const cached = await idbGet(key)
      if (cached && Date.now() - cached.timestamp < ttl) {
        state.data = cached.data
        state.loading = false
        // 后台再验证
        fetcher().then(d => idbSet(key, { data: d, timestamp: Date.now() }))
      } else {
        const data = await fetcher()
        state.data = data
        await idbSet(key, { data, timestamp: Date.now() })
      }
    } catch (e) {
      state.error = e
    } finally {
      state.loading = false
    }
  }

  watchEffect(() => {
    revalidate()
    if (refreshInterval > 0) {
      const id = setInterval(revalidate, refreshInterval)
      return () => clearInterval(id)
    }
  })

  return toRefs(state)
}
疑问 一句话答案 可展开细节
为什么选 IndexedDB 而不是 localStorage？ IndexedDB 支持 结构化克隆、容量大（≈ 用户磁盘 6–8 %），而 localStorage 只能存字符串且上限 5 MB。 可补充：localStorage 同步阻塞主线程；IndexedDB 异步不卡 UI。
怎么防止内存泄漏？ watchEffect 返回的清理函数在组件卸载时自动 clearInterval，同时每次请求新建 AbortController 可中断过期请求。 代码中已演示 return () => clearInterval(id)。
缓存击穿 / 并发请求怎么解决？ 利用 watchEffect 的响应式收集 + 单例 Promise 缓存；同 key 的并发 revalidate 会复用同一次网络请求。 可加 const promiseMap = new Map() 做请求去重。
TTL 到期后仍离线怎么办？ 捕获网络错误后 降级使用过期缓存，保证“可展示”而非白屏。 可在 catch 里 state.data = cached?.data 并打标记 isStale=true。
SSR 场景下如何预填充？ 服务端把数据写进 HTML 里的 __INITIAL_STATE__，客户端 hydrate 完成后 idbSet(key, data)，实现“秒开 + 后台更新”。 示例：if (import.meta.env.SSR) idbSet(...)。
